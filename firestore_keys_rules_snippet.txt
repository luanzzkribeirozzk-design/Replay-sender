// ReplayX — substituir somente o bloco match /keys/{keyId} existente.
// Mantenha os blocos atuais de pair_codes, pairings e transfers.
// O APK usa Firestore REST sem Firebase Auth; por isso este bloco limita a
// progressão dos slots, mas não substitui autenticação/App Check/backend.

match /keys/{keyId} {
  // Necessário para o login consultar a key pelo campo keyString.
  allow get, list: if true;

  // O aplicativo só pode avançar o uso da key. Nunca pode apagar, reduzir
  // devicesUsed ou limpar um slot já preenchido.
  allow update: if
    request.resource.data.diff(resource.data).affectedKeys().hasOnly([
      'slot1DeviceId', 'slot1Type', 'slot1Model', 'slot1UsedAt',
      'slot2DeviceId', 'slot2Type', 'slot2Model', 'slot2UsedAt',
      'devicesUsed', 'firstUsed'
    ])
    && request.resource.data.get('devicesUsed', 0) >= resource.data.get('devicesUsed', 0)
    && request.resource.data.get('devicesUsed', 0) <= 2
    && (resource.data.get('slot1DeviceId', '') == ''
        || request.resource.data.get('slot1DeviceId', '') == resource.data.get('slot1DeviceId', ''))
    && (resource.data.get('slot2DeviceId', '') == ''
        || request.resource.data.get('slot2DeviceId', '') == resource.data.get('slot2DeviceId', ''));

  allow create, delete: if false;
}
