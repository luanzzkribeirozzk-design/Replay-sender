// Substitua somente o bloco match /keys/{keyId} existente.
// Os campos antigos continuam iguais para não quebrar outros APKs.

match /keys/{keyId} {
  allow get, list: if true;
  allow create: if request.resource.data.keys().hasAll(['keyString','generatedBy','status','days'])
                && request.resource.data.status == 'active';
  allow update: if request.resource.data.diff(resource.data).affectedKeys()
                    .hasOnly([
                      'status', 'pausedAt', 'firstUsed', 'deviceId', 'deviceModel', 'lastIP', 'days',
                      'slot1DeviceId', 'slot1Type', 'slot1Model', 'slot1UsedAt',
                      'slot2DeviceId', 'slot2Type', 'slot2Model', 'slot2UsedAt',
                      'devicesUsed'
                    ])
                && request.resource.data.get('devicesUsed', 0) >= resource.data.get('devicesUsed', 0)
                && request.resource.data.get('devicesUsed', 0) <= 2
                && (resource.data.get('slot1DeviceId', '') == ''
                    || request.resource.data.get('slot1DeviceId', '') == resource.data.get('slot1DeviceId', ''))
                && (resource.data.get('slot2DeviceId', '') == ''
                    || request.resource.data.get('slot2DeviceId', '') == resource.data.get('slot2DeviceId', ''));
  // A ocupação exata dos slots não é imposta aqui para permitir a migração
                // gradual de documentos legados que ainda não possuem todos os campos.
  allow delete: if true;
}
